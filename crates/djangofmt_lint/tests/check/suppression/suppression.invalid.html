{# The formatter directive is not a lint suppression #} {# djangofmt:ignore #}
{# An empty rule list does not suppress #} {# djangofmt: ignore[] #}
{# Unknown rule matches nothing #} {# djangofmt: ignore[not-a-real-rule] #}
{# Rule mismatch: this silences a different rule than the one that fires #} {# djangofmt: ignore[empty-attr-value] #}
{# A comment after the node does not reach back to it #}
{# djangofmt: ignore[invalid-attr-value] #}
